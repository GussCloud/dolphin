# Dolphin relay zsh overlay wrapper
__dolphin_usable_zdotdir() {
  [[ -n "${1:-}" ]] || return 1
  # Dolphin's own dir, by marker file or by the shape older builds wrote.
  [[ "$1" != */shell-ready/zsh ]] || return 1
  [[ ! -f "$1/.dolphin-shell-wrapper" ]] || return 1
  # A directory holding no zsh startup file at all is not a config root,
  # whoever wrote it — and a stale value pointing at one would stop zsh from
  # ever reading the user's real .zshenv.
  local _dolphin_startup_file
  for _dolphin_startup_file in .zshenv .zshrc .zprofile .zlogin; do
    [[ -r "$1/$_dolphin_startup_file" ]] && return 0
  done
  return 1
}
if __dolphin_usable_zdotdir "${DOLPHIN_ORIG_ZDOTDIR:-}"; then
  builtin export ZDOTDIR="$DOLPHIN_ORIG_ZDOTDIR"
else
  builtin unset ZDOTDIR
fi
builtin unset DOLPHIN_ORIG_ZDOTDIR DOLPHIN_ZSHENV_SOURCE_DIR
builtin unfunction __dolphin_usable_zdotdir
builtin typeset -ga _dolphin_shell_features
_dolphin_shell_features=(${(s:,:)${DOLPHIN_SHELL_FEATURES:-}})
builtin unset DOLPHIN_SHELL_FEATURES
# Why DOLPHIN_HISTFILE is consumed HERE and not in the deferred hook: a user config
# that replaces precmd_functions wholesale drops the hook, and an exported value
# nothing will ever consume is then inherited by every child of this pane,
# including a nested Dolphin (#11146). Captured non-exported, it cannot escape.
builtin typeset -g _dolphin_histfile="${DOLPHIN_HISTFILE:-}"
builtin unset DOLPHIN_HISTFILE
__dolphin_has_feature() { (( ${_dolphin_shell_features[(Ie)$1]} )) }
__dolphin_has_feature identity && printf "\033]777;dolphin-shell-start:%s\007" "$$"
__dolphin_deferred_init() {
  # Why first: this body runs after the user's own config, so it would otherwise
  # inherit whatever options that config left set. Under NO_UNSET an unset
  # precmd_functions is fatal, and KSH_ARRAYS makes the 1-based feature lookup
  # drop whichever feature is listed first.
  builtin emulate -L zsh
  (( $+_dolphin_deferred_init_done )) && return 0
  builtin typeset -g _dolphin_deferred_init_done=1
  builtin typeset -g precmd_functions
  precmd_functions=(${precmd_functions:#__dolphin_deferred_init})
  if __dolphin_has_feature overlay; then
    # Why: remote startup files can re-export user defaults after relay spawn.
    [[ -n "${DOLPHIN_OPENCODE_CONFIG_DIR:-}" ]] && export OPENCODE_CONFIG_DIR="${DOLPHIN_OPENCODE_CONFIG_DIR}"
    [[ -n "${DOLPHIN_MIMOCODE_HOME:-}" ]] && export MIMOCODE_HOME="${DOLPHIN_MIMOCODE_HOME}"
    [[ -n "${DOLPHIN_REMOTE_CLI_BIN_DIR:-}" ]] && case ":$PATH:" in *:"${DOLPHIN_REMOTE_CLI_BIN_DIR}":*) ;; *) export PATH="${DOLPHIN_REMOTE_CLI_BIN_DIR}:$PATH" ;; esac
    # Why: OMP does not auto-load Dolphin's managed status extension; wrap only
    # interactive launch invocations so subcommands such as `omp config` keep
    # their normal argv shape.
    __dolphin_omp_should_skip_extension() {
      case "${1:-}" in
        'help'|'--help'|'-h'|'--version'|'-v') return 0 ;;
        '__complete'|'acp'|'agents'|'auth-broker'|'auth-gateway'|'bench'|'commit'|'completions'|'config'|'dry-balance'|'gallery'|'grep'|'grievances'|'install'|'join'|'models'|'plugin'|'read'|'say'|'search'|'setup'|'shell'|'ssh'|'stats'|'tiny-models'|'token'|'ttsr'|'update'|'usage'|'worktree'|'q'|'wt') return 0 ;;
      esac
      return 1
    }
    __dolphin_omp_cwd_is_usable() {
      local __dolphin_physical_cwd
      [[ -x . ]] || return 1
      if [[ -n "${PWD:-}" && -d "${PWD:-}" ]]; then
        [[ "${PWD}" -ef . ]]
      else
        # Why compare the path: shell builtins can print a cached path for a deleted cwd.
        __dolphin_physical_cwd="$(builtin pwd -P 2>/dev/null)" || return 1
        [[ -d "$__dolphin_physical_cwd" && "$__dolphin_physical_cwd" -ef . ]]
      fi
    }
    __dolphin_omp_invoke() {
      local __dolphin_use_extension="$1"
      shift
      if [[ $__dolphin_use_extension -eq 1 && -n "${DOLPHIN_OMP_STATUS_EXTENSION:-}" && -f "${DOLPHIN_OMP_STATUS_EXTENSION}" ]]; then
        if [[ "${1:-}" == "launch" ]]; then
          shift
          command omp launch --extension "${DOLPHIN_OMP_STATUS_EXTENSION}" "$@"
        else
          command omp --extension "${DOLPHIN_OMP_STATUS_EXTENSION}" "$@"
        fi
      else
        command omp "$@"
      fi
    }
    __dolphin_omp() {
      local __dolphin_use_extension=1
      __dolphin_omp_should_skip_extension "${1:-}" && __dolphin_use_extension=0
      if ! __dolphin_omp_cwd_is_usable; then
        local __dolphin_logical_cwd="${PWD:-${DOLPHIN_WORKTREE_PATH:-${DOLPHIN_ROOT_PATH:-}}}"
        # Why: a restored shell can retain the deleted directory inode after its path is recreated.
        (
          if [[ -z "$__dolphin_logical_cwd" ]]; then
            printf 'Dolphin: OMP cannot start because no terminal working directory is available. Open a new terminal in an existing directory.\n' >&2
            return 1
          fi
          if ! builtin cd -P -- "$__dolphin_logical_cwd" 2>/dev/null; then
            printf 'Dolphin: OMP cannot access the terminal working directory "%s". Open a new terminal in an existing directory.\n' "$__dolphin_logical_cwd" >&2
            return 1
          fi
          __dolphin_omp_invoke "$__dolphin_use_extension" "$@"
        )
      else
        __dolphin_omp_invoke "$__dolphin_use_extension" "$@"
      fi
    }
    if [[ -n "${DOLPHIN_OMP_STATUS_EXTENSION:-}" ]]; then
      # Why the function reserved word: it suppresses alias expansion of the name, which
      # an `alias omp` otherwise rewrites at parse time, aborting the rest of the file.
      function omp { __dolphin_omp "$@"; }
    fi
  fi
  if [[ -n "${_dolphin_histfile:-}" ]]; then
    HISTFILE="$_dolphin_histfile"
  fi
  if __dolphin_has_feature ready; then
    # Why: capture the prior zle-line-init so the marker chains to it. On a
    # re-source we are already the bound widget, so keep the function captured
    # the first time instead of clobbering it to empty (which would silently
    # drop the user's widget on every prompt after the second source). Only
    # user-defined widgets are chainable as plain functions; builtin/completion
    # forms (rare for zle-line-init) are left unchained.
    if [[ "${widgets[zle-line-init]:-}" == "user:__dolphin_prompt_mark" ]]; then
      :
    elif (( ${+widgets[zle-line-init]} )) && [[ "${widgets[zle-line-init]}" == user:* ]]; then
      __dolphin_prev_line_init_fn="${widgets[zle-line-init]#user:}"
    else
      __dolphin_prev_line_init_fn=""
    fi
    __dolphin_prompt_mark() {
      printf "\033]777;dolphin-shell-ready\007"
      # Why: call the prior hook as a plain function, not an aliased widget, so
      # $WIDGET stays zle-line-init for add-zle-hook-widget dispatchers.
      if [[ -n "${__dolphin_prev_line_init_fn:-}" ]]; then
        "${__dolphin_prev_line_init_fn}" "$@"
      fi
    }
    zle -N zle-line-init __dolphin_prompt_mark
  fi
  builtin unset _dolphin_shell_features _dolphin_histfile
  builtin unfunction __dolphin_deferred_init __dolphin_has_feature
}
{
  builtin typeset _dolphin_user_zshenv="${ZDOTDIR-$HOME}/.zshenv"
  [[ ! -r "$_dolphin_user_zshenv" ]] || builtin source -- "$_dolphin_user_zshenv"
} always {
  builtin unset _dolphin_user_zshenv
  builtin typeset -ag precmd_functions
  (( ${precmd_functions[(Ie)__dolphin_deferred_init]} )) || precmd_functions+=(__dolphin_deferred_init)
}
